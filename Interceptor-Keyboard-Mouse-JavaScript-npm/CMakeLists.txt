cmake_minimum_required(VERSION 3.24)
project(ProductNodeSDK LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(FAMILY_BRIDGE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../Interceptor-Keyboard-Mouse-C-ffi" CACHE PATH "Product C ABI source")
set(NODE_HEADERS "" CACHE PATH "Node-API headers")
set(NODE_LIBRARY "" CACHE FILEPATH "Node import library")
add_subdirectory("${FAMILY_BRIDGE_SOURCE}" bridge)
set_target_properties(InterceptorKeyboardMouseBridge PROPERTIES OUTPUT_NAME InterceptorKeyboardMouseBridge)
add_library(interceptor_keyboard_mouse_sdk_node MODULE native.cpp DelayLoad.cpp)
set_target_properties(interceptor_keyboard_mouse_sdk_node PROPERTIES PREFIX "" SUFFIX ".node")
target_include_directories(interceptor_keyboard_mouse_sdk_node PRIVATE "${FAMILY_BRIDGE_SOURCE}" "${NODE_HEADERS}")
target_compile_features(interceptor_keyboard_mouse_sdk_node PRIVATE cxx_std_17)
target_compile_definitions(interceptor_keyboard_mouse_sdk_node PRIVATE NAPI_VERSION=8 NODE_GYP_MODULE_NAME=interceptor_keyboard_mouse_sdk_node)
target_link_libraries(interceptor_keyboard_mouse_sdk_node PRIVATE InterceptorKeyboardMouseBridge "${NODE_LIBRARY}" delayimp)
target_link_options(interceptor_keyboard_mouse_sdk_node PRIVATE /DELAYLOAD:node.exe)
if(FAMILY_TEST_BACKEND)
 target_compile_definitions(interceptor_keyboard_mouse_sdk_node PRIVATE ICB_TEST_BACKEND)
endif()
